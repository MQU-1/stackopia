const express = require('express');
const cors = require('cors');
const Anthropic = require('@anthropic-ai/sdk');

const app = express();
app.use(cors());
app.use(express.json());
app.use(express.static('public'));

const anthropic = new Anthropic({
  apiKey: process.env.ANTHROPIC_API_KEY,
});

app.post('/api/ai-proxy', async (req, res) => {
  try {
    const data = req.body;
    let systemPrompt = "You are a financial assistant. You must respond strictly in valid JSON format, with no markdown formatting or conversational text.";
    let userMessage = "";

    if (data.type === 'spend') {
      userMessage = `
        Evaluate this purchase: ${data.item} for JD ${data.amount}. Context: ${data.ctx}.
        Return ONLY a JSON object with:
        - "score": A number from 1-10.
        - "verdict": A short string (e.g., "Great Investment", "Impulse Buy", "Terrible Idea").
        - "insight": A brutally honest 2-sentence explanation.
      `;
    } 
    else if (data.type === 'invest') {
      userMessage = `
        Create a simple investment plan for JD ${data.amount} with a "${data.risk}" risk profile.
        Return ONLY a JSON object with:
        - "summary": A 1-sentence overview.
        - "allocations": An array of objects, each containing "label" (string), "percent" (number), and "color" (a hex code string like "#C8F135"). The percents must add up to 100.
        - "advice": A 2-sentence piece of practical advice.
      `;
    }
    else if (data.type === 'price') {
      userMessage = `
        Estimate local Amman and online prices for: ${data.query}.
        Return ONLY a JSON object with:
        - "title": "Price check for ${data.query}"
        - "stores": An array of 3 objects, each containing "name", "type" (e.g., "Local Tech Store", "Amazon"), "note" (e.g., "Includes warranty"), and "price" (a number representing JD). Order from cheapest to most expensive.
        - "tip": 1 sentence of buying advice.
      `;
    } 
    else {
      return res.status(400).json({ error: "Invalid request type" });
    }

    const msg = await anthropic.messages.create({
      model: "claude-3-haiku-20240307",
      max_tokens: 500,
      temperature: 0.7,
      system: systemPrompt,
      messages: [{ role: "user", content: userMessage }],
    });

    const aiText = msg.content[0].text;
    const aiJson = JSON.parse(aiText);

    res.json(aiJson);

  } catch (error) {
    console.error("AI Service Error:", error);
    res.status(500).json({ error: "Failed to generate AI response." });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`✅ Secure AI Proxy running on port ${PORT}`);
});

document.addEventListener('DOMContentLoaded', () => {
  let selectedRisk = 'safe';
  document.querySelectorAll('[data-nav]').forEach(el => {
    el.addEventListener('click', (e) => { showPage(e.currentTarget.getAttribute('data-nav')); });
  });
  document.querySelectorAll('.risk-opt').forEach(el => {
    el.addEventListener('click', (e) => {
      selectedRisk = e.currentTarget.getAttribute('data-risk');
      document.querySelectorAll('.risk-opt').forEach(r => r.classList.remove('sel'));
      e.currentTarget.classList.add('sel');
    });
  });
  document.getElementById('btn-rate-spend')?.addEventListener('click', rateSpend);
  document.getElementById('btn-get-invest')?.addEventListener('click', getInvest);
  document.getElementById('btn-hunt-prices')?.addEventListener('click', huntPrices);

  function showPage(id) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.querySelectorAll('.nav-tab').forEach(t => t.classList.remove('active'));
    document.getElementById('page-' + id).classList.add('active');
    const activeTab = document.querySelector(`.nav-tab[data-nav="${id}"]`);
    if(activeTab) activeTab.classList.add('active');
    if (id === 'savings') { setTimeout(() => { document.getElementById('prog1').style.width = '67%'; }, 80); }
  }

  async function callAIBackend(promptData) {
    const res = await fetch('/api/ai-proxy', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(promptData)
    });
    if(!res.ok) throw new Error("Backend AI service error");
    return await res.json();
  }

  async function rateSpend() {
    const item = document.getElementById('spend-item').value.trim();
    const amount = document.getElementById('spend-amount').value.trim();
    const ctx = document.getElementById('spend-context').value.trim();
    if (!item || !amount) { alert('Please fill in the purchase and amount.'); return; }
    const loading = document.getElementById('spend-loading');
    const result = document.getElementById('rating-result');
    loading.classList.add('show'); result.style.display = 'none';
    try {
      const r = await callAIBackend({ type: 'spend', item, amount, ctx });
      loading.classList.remove('show');
      document.getElementById('score-num').textContent = r.score + '/10';
      const bar = document.getElementById('score-bar');
      bar.style.background = r.score >= 7 ? 'var(--accent2)' : r.score >= 4 ? 'var(--accent3)' : 'var(--red)';
      setTimeout(() => bar.style.width = (r.score * 10) + '%', 80);
      const badge = document.getElementById('verdict-badge');
      badge.textContent = r.verdict;
      badge.className = 'verdict-badge ' + (r.score >= 7 ? 'vg' : r.score >= 4 ? 'vo' : 'vb');
      document.getElementById('insight-text').textContent = r.insight;
      result.style.display = 'block';
    } catch(e) {
      loading.classList.remove('show');
      alert('Error calling AI proxy.');
    }
  }

  async function getInvest() {
    const amt = document.getElementById('invest-amt').value.trim();
    if (!amt) { alert('Enter your saved amount first.'); return; }
    const loading = document.getElementById('invest-loading');
    const result = document.getElementById('invest-result');
    loading.classList.add('show'); result.style.display = 'none';
    try {
      const r = await callAIBackend({ type: 'invest', amount: amt, risk: selectedRisk });
      loading.classList.remove('show');
      document.getElementById('invest-summary').textContent = r.summary;
      const bar = document.getElementById('alloc-bar');
      const legend = document.getElementById('invest-legend');
      bar.innerHTML = ''; legend.innerHTML = '';
      r.allocations.forEach(a => {
        const seg = document.createElement('div');
        seg.className = 'alloc-seg'; seg.style.width = a.percent + '%'; seg.style.background = a.color;
        bar.appendChild(seg);
        const li = document.createElement('div');
        const colorBox = document.createElement('span'); colorBox.className = 'legend-dot'; colorBox.style.background = a.color;
        const percentStrong = document.createElement('strong'); percentStrong.style.color = 'var(--text)'; percentStrong.textContent = a.percent + '%';
        li.appendChild(colorBox); li.appendChild(document.createTextNode(a.label + ' ')); li.appendChild(percentStrong);
        legend.appendChild(li);
      });
      document.getElementById('invest-advice').textContent = r.advice;
      result.style.display = 'block';
    } catch(e) {
      loading.classList.remove('show');
      alert('Error calling AI proxy.');
    }
  }

  async function huntPrices() {
    const q = document.getElementById('price-q').value.trim();
    if (!q) { alert('Enter a product name first.'); return; }
    const loading = document.getElementById('price-loading');
    const results = document.getElementById('price-results');
    loading.classList.add('show'); results.style.display = 'none';
    try {
      const r = await callAIBackend({ type: 'price', query: q });
      loading.classList.remove('show');
      document.getElementById('price-title').textContent = r.title;
      const list = document.getElementById('price-list');
      list.innerHTML = '';
      r.stores.forEach((s, i) => {
        const row = document.createElement('div');
        row.className = 'price-row' + (i === 0 ? ' best' : '');
        const storeDiv = document.createElement('div');
        storeDiv.className = 'price-store';
        storeDiv.textContent = s.name;
        if (i === 0) {
          const badge = document.createElement('span'); badge.className = 'best-badge'; badge.textContent = '⚡ Best Price';
          storeDiv.appendChild(badge);
        }
        const typeDiv = document.createElement('div'); typeDiv.className = 'price-type'; typeDiv.textContent = s.type + ' · ' + s.note;
        const leftCol = document.createElement('div'); leftCol.appendChild(storeDiv); leftCol.appendChild(typeDiv);
        const rightCol = document.createElement('div'); rightCol.className = 'price-amt'; rightCol.style.color = i === 0 ? 'var(--accent)' : 'var(--text)';
        rightCol.textContent = 'JD ' + s.price;
        row.appendChild(leftCol); row.appendChild(rightCol); list.appendChild(row);
      });
      document.getElementById('price-tip').textContent = '💡 ' + r.tip;
      results.style.display = 'block';
    } catch(e) {
      loading.classList.remove('show');
      alert('Error calling AI proxy.');
    }
  }
});
